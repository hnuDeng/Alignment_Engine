class MultiObjectivePlanner:
    def __init__(self, planner, objectives):
        self.planner = planner
        self.objectives = objectives
    
    def plan(self, task, scene_description, preferences):
        # 生成多个候选规划
        candidates = self.generate_candidates(task, scene_description)
        
        # 评估每个候选
        scores = []
        for candidate in candidates:
            score = self.evaluate(candidate, preferences)
            scores.append(score)
        
        # 选择最佳候选
        best_candidate = candidates[argmax(scores)]
        
        return best_candidate
    
    def evaluate(self, candidate, preferences):
        # 评估多个目标
        scores = {}
        for objective in self.objectives:
            scores[objective.name] = objective.evaluate(candidate)
        
        # 加权求和
        total_score = sum(preferences[obj] * scores[obj] for obj in scores)
        
        return total_score
```

class FaultTolerantPlanner:
    def __init__(self, planner, fault_detector, recovery_planner):
        self.planner = planner
        self.fault_detector = fault_detector
        self.recovery_planner = recovery_planner
    
    def plan(self, task, scene_description):
        # 生成规划
        plan = self.planner.plan(task, scene_description)
        
        # 添加故障检测点
        plan_with_checkpoints = self.add_checkpoints(plan)
        
        return plan_with_checkpoints
    
    def execute(self, plan):
        for step in plan:
            # 执行步骤
            result = self.execute_step(step)
            
            # 检测故障
            if self.fault_detector.detect(result):
                # 诊断故障
                fault = self.fault_detector.diagnose(result)
                
                # 恢复
                recovery_plan = self.recovery_planner.plan(fault)
                self.execute(recovery_plan)
```
