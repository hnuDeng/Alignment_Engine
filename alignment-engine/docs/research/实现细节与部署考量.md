```python
# 系统架构
class HierarchicalLLMPlanner:
    def __init__(self):
        # LLM接口
        self.llm = LLMInterface()
        
        # 任务分解器
        self.task_decomposer = TaskDecomposer(self.llm)
        
        # 原语库
        self.primitive_library = PrimitiveLibrary()
        
        # 原语选择器
        self.primitive_selector = PrimitiveSelector(self.llm, self.primitive_library)
        
        # 参数优化器
        self.parameter_optimizer = ParameterOptimizer()
        
        # 轨迹生成器
        self.trajectory_generator = TrajectoryGenerator()
        
        # 物理验证器
        self.physics_verifier = PhysicsVerifier()
        
        # 反馈调整器
        self.feedback_adjuster = FeedbackAdjuster()
        
        # 重规划器
        self.replanner = Replanner(self.llm)
    
    def plan(self, instruction, scene_description, current_state):
        """
        主规划流程
        """
        max_iterations = 5
        
        for iteration in range(max_iterations):
            # Step 1: 任务分解
            subtasks = self.task_decomposer.decompose(instruction, scene_description)
            
            # Step 2: 原语选择
            primitives = []
            for subtask in subtasks:
                primitive = self.primitive_selector.select(subtask)
                primitives.append(primitive)
            
            # Step 3: 参数优化
            optimized_primitives = []
            for primitive in primitives:
                optimized = self.parameter_optimizer.optimize(primitive, current_state)
                optimized_primitives.append(optimized)
            
            # Step 4: 轨迹生成
            trajectory = self.trajectory_generator.generate(optimized_primitives)
            
            # Step 5: 物理验证
            verification_result = self.physics_verifier.verify(trajectory)
            
            # Step 6: 检查是否通过
            if verification_result.passed:
                return trajectory, "Success"
            
            # Step 7: 反馈调整
            feedback = self.feedback_adjuster.adjust(verification_result)
            
            # Step 8: 重规划
            instruction = self.replanner.replan(instruction, feedback)
        
        return None, "Max iterations reached"
```

### 2.3 参数优化器实现

```python
class ParameterOptimizer:
    def __init__(self, learning_rate=0.01, max_iterations=100):
        self.learning_rate = learning_rate
        self.max_iterations = max_iterations
    
    def optimize(self, primitive, current_state):
        """
        优化原语参数
        """
        # 获取初始参数
        params = primitive["parameters"].copy()
        
        # 定义目标函数
        def objective(params):
            # 生成轨迹
            trajectory = primitive["trajectory_generator"](params)
            
            # 评估轨迹
            score = self.evaluate_trajectory(trajectory, current_state)
            
            return -score  # 最小化负分数
        
        # 使用梯度下降优化
        best_params = params.copy()
        best_score = objective(params)
        
        for iteration in range(self.max_iterations):
            # 计算梯度
            grad = self.compute_gradient(objective, best_params)
            
            # 更新参数
            new_params = {
                k: v - self.learning_rate * grad[k] 
                for k, v in best_params.items()
            }
            
            # 评估新参数
            new_score = objective(new_params)
            
            # 更新最佳参数
            if new_score > best_score:
                best_params = new_params
                best_score = new_score
        
        return best_params
```

### 2.4 物理验证器实现

```python
class PhysicsVerifier:
    def __init__(self, simulator):
        self.simulator = simulator
    
    def verify(self, trajectory):
        """
        验证轨迹物理可行性
        """
        results = {
            "collision_free": True,
            "joint_limits_ok": True,
            "velocity_limits_ok": True,
            "passed": True,
            "details": []
        }
        
        # 检查碰撞
        collision_result = self.check_collision(trajectory)
        if not collision_result["passed"]:
            results["collision_free"] = False
            results["details"].append(collision_result["details"])
        
        # 检查关节限位
        joint_result = self.check_joint_limits(trajectory)
        if not joint_result["passed"]:
            results["joint_limits_ok"] = False
            results["details"].append(joint_result["details"])
        
        # 检查速度限制
        velocity_result = self.check_velocity_limits(trajectory)
        if not velocity_result["passed"]:
            results["velocity_limits_ok"] = False
            results["details"].append(velocity_result["details"])
        
        # 总体结果
        results["passed"] = all([
            results["collision_free"],
            results["joint_limits_ok"],
            results["velocity_limits_ok"]
        ])
        
        return results
```

```python
        # 检查缓存
```

```python
class PlannerMetrics:
    def __init__(self):
        self.metrics = {
            "total_plans": 0,
            "successful_plans": 0,
            "failed_plans": 0,
            "average_planning_time": 0,
            "average_iterations": 0
        }
    
    def update(self, status, planning_time, iterations):
        self.metrics["total_plans"] += 1
        
        if status == "Success":
            self.metrics["successful_plans"] += 1
        else:
            self.metrics["failed_plans"] += 1
        
        # 更新平均时间
        n = self.metrics["total_plans"]
        self.metrics["average_planning_time"] = (
            (self.metrics["average_planning_time"] * (n - 1) + planning_time) / n
        )
        
        # 更新平均迭代次数
        self.metrics["average_iterations"] = (
            (self.metrics["average_iterations"] * (n - 1) + iterations) / n
        )
    
    def report(self):
        return {
            "success_rate": self.metrics["successful_plans"] / max(1, self.metrics["total_plans"]),
            "average_planning_time": self.metrics["average_planning_time"],
            "average_iterations": self.metrics["average_iterations"]
        }
```
